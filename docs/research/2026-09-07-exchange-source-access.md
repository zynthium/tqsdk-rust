```bash

  'https://www.czce.com.cn/cn/rootfiles/2024/02/22/1708569360987406-1708569361004511.pdf' \
  -o /tmp/czce-january-2024-public.pdf

sha256sum /tmp/czce-january-2024-public.pdf
```
